<script>
  let isSyncing = false;

  window.onload = function() {
    let today = new Date().toISOString().split('T')[0];
    document.getElementById('contraDate').value = today;
    document.getElementById('fromDate').value = today;
    document.getElementById('toDate').value = today;
    // Page load par enter press ki shart ke mutabiq auto fetch nahi hoga, user enter press karega
  };

  function showLoader(show) {
    document.getElementById('loader').style.display = show ? 'block' : 'none';
  }

  // Sirf Enter key press hone par working start hogi
  function handleDateEnter(event) {
    if (event.key === 'Enter') {
      let fromDate = document.getElementById('fromDate').value;
      let toDate = document.getElementById('toDate').value;
      
      if (fromDate && toDate) {
        fetchRangeSummary(fromDate, toDate);
      } else {
        alert("Baraye meherbani From Date aur To Date dono select karein!");
      }
    }
  }

  function submitContra() {
    let data = {
      date: document.getElementById('contraDate').value,
      type: document.getElementById('contraType').value,
      amount: document.getElementById('contraAmount').value,
      description: document.getElementById('contraDesc').value,
      enteredBy: "Admin"
    };

    if (!data.date || !data.amount) {
      alert("Date and Amount are required for Contra Voucher!");
      return;
    }

    showLoader(true);
    google.script.run
      .withSuccessHandler(function(res) {
        showLoader(false);
        if (res.success) {
          alert("Contra Voucher Saved Successfully!\nContra No: " + res.contraNo);
          clearContraForm();
        } else {
          alert("Error: " + res.error);
        }
      })
      .withFailureHandler(function(err) {
        showLoader(false);
        alert("Server Error: " + err.message);
      })
      .saveContraVoucher(data);
  }

  function clearContraForm() {
    document.getElementById('contraAmount').value = '';
    document.getElementById('contraDesc').value = '';
    let today = new Date().toISOString().split('T')[0];
    document.getElementById('contraDate').value = today;
  }

  function calculateDifference() {
    let opening = parseFloat(document.getElementById('openingCash').value) || 0;
    let cashIn = parseFloat(document.getElementById('totalCashIn').value) || 0;
    let cashOut = parseFloat(document.getElementById('totalCashOut').value) || 0;
    
    let closing = (opening + cashIn) - cashOut;
    document.getElementById('calculatedClosing').value = closing;
  }

  function submitClosing() {
    let data = {
      fromDate: document.getElementById('fromDate').value,
      toDate: document.getElementById('toDate').value,
      openingCash: document.getElementById('openingCash').value,
      totalCashIn: document.getElementById('totalCashIn').value,
      totalCashOut: document.getElementById('totalCashOut').value,
      closingCash: document.getElementById('calculatedClosing').value,
      status: "Closed"
    };

    showLoader(true);
    google.script.run
      .withSuccessHandler(function(res) {
        showLoader(false);
        if (res.success) {
          alert("Daily/Range Closing Saved Successfully!");
        } else {
          alert("Error: " + res.error);
        }
      })
      .withFailureHandler(function(err) {
        showLoader(false);
        alert("Server Error: " + err.message);
      })
      .saveDailyClosing(data);
  }

  function fetchRangeSummary(fromDate, toDate) {
    document.getElementById('loadingOverlay').style.display = 'block';

    google.script.run
      .withSuccessHandler(function(res) {
        document.getElementById('loadingOverlay').style.display = 'none';
        if (res.success) {
          document.getElementById('openingCash').value = res.openingCash;
          document.getElementById('totalCashIn').value = res.totalCashIn;
          document.getElementById('totalCashOut').value = res.totalCashOut;
          calculateDifference();
        }
      })
      .withFailureHandler(function(err) {
        document.getElementById('loadingOverlay').style.display = 'none';
        console.error(err.message);
      })
      .getRangeSummaryData(fromDate, toDate);
  }

  function runSync() {
    isSyncing = false;
    document.getElementById('loadingOverlay').style.display = 'block';
    document.getElementById('stopSyncBtn').style.display = 'inline-block';

    google.script.run
      .withSuccessHandler(function(res) {
        document.getElementById('loadingOverlay').style.display = 'none';
        document.getElementById('stopSyncBtn').style.display = 'none';
        if (res.success) {
          alert("Sync Completed Successfully! Total Added: " + res.count);
          let fromDate = document.getElementById('fromDate').value;
          let toDate = document.getElementById('toDate').value;
          if(fromDate && toDate) fetchRangeSummary(fromDate, toDate);
        }
      })
      .withFailureHandler(function(err) {
        document.getElementById('loadingOverlay').style.display = 'none';
        document.getElementById('stopSyncBtn').style.display = 'none';
        alert("Error: " + err.message);
      })
      .syncReceiptsToCashBook();
  }

  function stopSync() {
    isSyncing = true;
    document.getElementById('loadingOverlay').style.display = 'none';
    document.getElementById('stopSyncBtn').style.display = 'none';
    alert("Sync process stopped by user.");
  }
</script>
