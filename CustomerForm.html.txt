<script>
  // Page load hotay hi next account number automatically fetch ho jaye ga
  window.onload = function() {
    loadNextAccountNo();
  };

  function loadNextAccountNo() {
    google.script.run
      .withSuccessHandler(function(res) {
        if (res && res.success) {
          document.getElementById('accNo').value = res.nextAccNo;
        }
      })
      .getNextAccountNumber();
  }

  function showLoader(show) {
    document.getElementById('loader').style.display = show ? 'block' : 'none';
  }

  function saveCustomer() {
    let data = {
      accountTitle: document.getElementById('accTitle').value.trim(),
      salesman: document.getElementById('salesman').value.trim(),
      subSalesman: document.getElementById('subSalesman').value.trim(),
      mobileNo: document.getElementById('mobileNo').value.trim(),
      address: document.getElementById('address').value.trim(),
      department: document.getElementById('department').value.trim(),
      salary: document.getElementById('salary').value.trim(),
      designation: document.getElementById('designation').value.trim(),
      
      g1Name: document.getElementById('g1Name').value.trim(),
      g1Mobile: document.getElementById('g1Mobile').value.trim(),
      g1Dept: document.getElementById('g1Dept').value.trim(),
      g1Desig: document.getElementById('g1Desig').value.trim(),

      g2Name: document.getElementById('g2Name').value.trim(),
      g2Mobile: document.getElementById('g2Mobile').value.trim(),
      g2Dept: document.getElementById('g2Dept').value.trim(),
      g2Desig: document.getElementById('g2Desig').value.trim()
    };

    if (!data.accountTitle) {
      alert("Account Title (Customer Name) is required!");
      return;
    }

    showLoader(true);
    google.script.run
      .withSuccessHandler(function(res) {
        showLoader(false);
        if (res && res.success) {
          alert("Customer Saved Successfully!");
          clearForm();
          loadNextAccountNo(); // Agla account number foran generate ho kar aa jaye ga
        } else {
          alert("Error: " + (res ? res.error : "Unknown error"));
        }
      })
      .withFailureHandler(function(err) {
        showLoader(false);
        alert("Server Error: " + err.message);
      })
      .saveNewCustomer(data);
  }

  function clearForm() {
    // Sab inputs clear hon gi lekin accNo apni jagah safe rahega taake next number na disturb ho
    const inputs = document.querySelectorAll('input:not(#accNo)');
    inputs.forEach(input => input.value = '');
  }
</script>
