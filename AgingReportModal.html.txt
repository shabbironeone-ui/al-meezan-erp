  <script>
    let globalData = [];
    let rawAccountsList = [];
    let currentRenderedRows = [];

    window.onload = function() {
      setLoading(true);
      google.script.run
        .withSuccessHandler(function(response) {
          setLoading(false);
          document.getElementById('loading').style.display = 'none';
          globalData = response.data;
          rawAccountsList = response.filters.accountsObj || [];
          populateDropdown('filterSalesman', response.filters.salesmen);
          populateDropdown('filterSubSalesman', response.filters.subSalesmen);
          populateAccountDropdowns(rawAccountsList);
          renderTable(globalData);
          setTimeout(makeColumnsResizable, 300);
        })
        .withFailureHandler(function(err) {
          setLoading(false);
          document.getElementById('loading').innerHTML = '<span style="color:red;">Error: ' + err.message + '</span>';
        })
        .getAgingReportData();

      document.addEventListener('click', function(e) {
        if (!e.target.closest('th')) {
          document.querySelectorAll('.suggestions-list').forEach(list => list.style.display = 'none');
        }
      });
    };

    function setLoading(isWaiting) {
      if (isWaiting) document.body.classList.add('waiting');
      else document.body.classList.remove('waiting');
    }

    function toggleColSearch(event, colIndex) {
      let th = event.currentTarget;
      let input = th.querySelector('.col-search');
      if (input) {
        if (input.style.display === 'block') {
          input.style.display = 'none';
          input.value = '';
          th.querySelector('.suggestions-list').style.display = 'none';
          filterTableColumns();
        } else {
          input.style.display = 'block';
          input.focus();
        }
      }
    }

    function getColumnUniqueValues(colIndex) {
      let valuesSet = new Set();
      let table = document.getElementById('reportTable');
      let trs = table.getElementsByTagName('tbody')[0].getElementsByTagName('tr');
      for (let i = 0; i < trs.length; i++) {
        if (trs[i].classList.contains('footer-total')) continue;
        let tdArray = trs[i].getElementsByTagName('td');
        if (tdArray[colIndex]) {
          let val = (tdArray[colIndex].textContent || tdArray[colIndex].innerText).trim();
          if (val) valuesSet.add(val);
        }
      }
      return Array.from(valuesSet).sort();
    }

    function onColSearchInput(event, colIndex) {
      event.stopPropagation();
      let input = event.target;
      let filterVal = input.value.toLowerCase();
      let wrapper = input.parentElement;
      let suggestionsBox = wrapper.querySelector('.suggestions-list');
      filterTableColumns();
      let uniqueVals = getColumnUniqueValues(colIndex);
      let matched = uniqueVals.filter(v => v.toLowerCase().includes(filterVal));
      suggestionsBox.innerHTML = '';
      if (matched.length > 0 && filterVal.length > 0) {
        matched.forEach(item => {
          let div = document.createElement('div');
          div.textContent = item;
          div.onclick = function(e) {
            e.stopPropagation();
            input.value = item;
            suggestionsBox.style.display = 'none';
            filterTableColumns();
          };
          suggestionsBox.appendChild(div);
        });
        suggestionsBox.style.display = 'block';
      } else {
        suggestionsBox.style.display = 'none';
      }
    }

    function filterTableColumns() {
      let table = document.getElementById('reportTable');
      let tbody = table.getElementsByTagName('tbody')[0];
      let trs = tbody.getElementsByTagName('tr');
      let inputs = table.querySelectorAll('th .col-search');
      
      let tSales = 0, tRec = 0, tBal = 0, tMonthlyInst = 0, t0_30 = 0, t31_60 = 0, t61_90 = 0, t91_plus = 0;

      for (let i = 0; i < trs.length; i++) {
        if (trs[i].classList.contains('footer-total')) continue;
        let tdArray = trs[i].getElementsByTagName('td');
        let showRow = true;
        for (let j = 0; j < inputs.length; j++) {
          let filterVal = inputs[j].value.toLowerCase();
          if (filterVal && tdArray[j]) {
            let textValue = tdArray[j].textContent || tdArray[j].innerText;
            if (textValue.toLowerCase().indexOf(filterVal) === -1) {
              showRow = false;
              break;
            }
          }
        }
        trs[i].style.display = showRow ? '' : 'none';

        if (showRow) {
          let originalRowData = currentRenderedRows[i];
          if (originalRowData) {
            tSales += originalRowData.totalSales;
            tRec += originalRowData.totalReceived;
            tBal += originalRowData.balance;

            // Only add to installment/aging totals if balance outstanding > 0
            if (originalRowData.balance > 0) {
              tMonthlyInst += originalRowData.monthlyInst;
              t0_30 += originalRowData.b0;
              t31_60 += originalRowData.b31;
              t61_90 += originalRowData.b61;
              t91_plus += originalRowData.b91;
            }
          }
        }
      }

      let totalTr = tbody.querySelector('.footer-total');
      if (totalTr) {
        totalTr.innerHTML = `
          <td colspan="7" style="text-align:right;">TOTAL:</td>
          <td class="number">${tSales.toLocaleString()}</td>
          <td class="number">${tRec.toLocaleString()}</td>
          <td class="number">${tBal.toLocaleString()}</td>
          <td></td>
          <td class="number">${tMonthlyInst ? tMonthlyInst.toLocaleString() : ''}</td>
          <td class="number">${t0_30 ? t0_30.toLocaleString() : ''}</td>
          <td class="number">${t31_60 ? t31_60.toLocaleString() : ''}</td>
          <td class="number">${t61_90 ? t61_90.toLocaleString() : ''}</td>
          <td class="number">${t91_plus ? t91_plus.toLocaleString() : ''}</td>
        `;
      }
    }

    function populateDropdown(elementId, items) {
      let select = document.getElementById(elementId);
      select.innerHTML = '<option value="">All ' + (elementId.includes('Salesman') ? 'Salesmen' : 'Accounts') + '</option>';
      if(elementId === 'filterSalesman') select.innerHTML = '<option value="">All Salesmen</option>';
      if(elementId === 'filterSubSalesman') select.innerHTML = '<option value="">All Sub Salesmen</option>';
      items.forEach(item => {
        let opt = document.createElement('option');
        opt.value = item;
        opt.textContent = item;
        select.appendChild(opt);
      });
    }

    function populateAccountDropdowns(accounts) {
      let fromSelect = document.getElementById('filterFromAccount');
      let toSelect = document.getElementById('filterToAccount');
      fromSelect.innerHTML = '<option value="">Start Account</option>';
      toSelect.innerHTML = '<option value="">End Account</option>';
      accounts.forEach(acc => {
        let opt1 = document.createElement('option');
        opt1.value = acc.accNo;
        opt1.textContent = acc.accNo + " - " + acc.accTitle;
        fromSelect.appendChild(opt1);
        let opt2 = document.createElement('option');
        opt2.value = acc.accNo;
        opt2.textContent = acc.accNo + " - " + acc.accTitle;
        toSelect.appendChild(opt2);
      });
    }

    function applyFilters() {
      setLoading(true);
      setTimeout(() => {
        let salesFilter = document.getElementById('filterSalesman').value;
        let subSalesFilter = document.getElementById('filterSubSalesman').value;
        let fromAcc = document.getElementById('filterFromAccount').value;
        let toAcc = document.getElementById('filterToAccount').value;
        let fromDateVal = document.getElementById('fromDate').value;
        let toDateVal = document.getElementById('toDate').value;
        let balOperator = document.getElementById('balanceOperator').value;
        let balValue = parseFloat(document.getElementById('balanceValue').value) || 0;

        let filtered = globalData.filter(row => {
          let matchSales = !salesFilter || row.salesman === salesFilter;
          let matchSubSales = !subSalesFilter || row.subSalesman === subSalesFilter;
          let matchAccountRange = true;
          let currentAccNum = String(row.accNo).trim();
          if (fromAcc && currentAccNum < fromAcc) matchAccountRange = false;
          if (toAcc && currentAccNum > toAcc) matchAccountRange = false;

          let matchDate = true;
          if (row.deliveryDate) {
            let rowDate = row.deliveryDate;
            if (fromDateVal && rowDate < fromDateVal) matchDate = false;
            if (toDateVal && rowDate > toDateVal) matchDate = false;
          } else if (fromDateVal || toDateVal) { matchDate = false; }

          let matchBalance = true;
          let rowBalance = parseFloat(row.balance) || 0;
          if (balOperator === 'gt') matchBalance = rowBalance > balValue;
          else if (balOperator === 'lt') matchBalance = rowBalance < balValue;
          else if (balOperator === 'eq') matchBalance = rowBalance === balValue;
          else if (balOperator === 'gt_zero') matchBalance = rowBalance > 0;

          return matchSales && matchSubSales && matchAccountRange && matchDate && matchBalance;
        });

        renderTable(filtered);
        setLoading(false);
      }, 50);
    }

    function resetFilters() {
      document.getElementById('filterSalesman').value = '';
      document.getElementById('filterSubSalesman').value = '';
      document.getElementById('filterFromAccount').value = '';
      document.getElementById('filterToAccount').value = '';
      document.getElementById('fromDate').value = '';
      document.getElementById('toDate').value = '';
      document.getElementById('balanceOperator').value = 'all';
      document.getElementById('balanceValue').value = '';
      renderTable(globalData);
    }

    function renderTable(data) {
      let table = document.getElementById('reportTable');
      let tbody = document.getElementById('tableBody');
      tbody.innerHTML = '';
      table.style.display = 'table';
      
      currentRenderedRows = [];

      if(data.length === 0) {
        tbody.innerHTML = '<tr><td colspan="16" style="text-align:center; padding: 20px;">No records match the selected filters.</td></tr>';
        return;
      }
      
      let tSales = 0, tRec = 0, tBal = 0, tMonthlyInst = 0, t0_30 = 0, t31_60 = 0, t61_90 = 0, t91_plus = 0;
      
      data.forEach(function(row) {
        tSales += row.totalSales;
        tRec += row.totalReceived;
        tBal += row.balance;

        let outstandingBal = parseFloat(row.balance) || 0;
        let monthlyInst = parseFloat(row.monthlyInst) || 0;
        let b0 = 0, b31 = 0, b61 = 0, b91 = 0;

        if (outstandingBal > 0 && monthlyInst > 0 && row.deliveryDate) {
          let dDate = new Date(row.deliveryDate);
          let startYear = dDate.getFullYear();
          let startMonth = dDate.getMonth() + 1;
          
          if (startMonth > 11) {
            startMonth = 0;
            startYear++;
          }
          let targetDate = new Date(startYear, startMonth, 1);
          let todayObj = new Date();

          let remainingBal = outstandingBal;
          let m = 0;
          while (remainingBal > 0) {
            let instDueDate = new Date(targetDate.getFullYear(), targetDate.getMonth() + m, targetDate.getDate());
            let diffTime = todayObj - instDueDate;
            let diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24));

            let instAmt = Math.min(monthlyInst, remainingBal);

            if (diffDays >= 0) {
              if (diffDays <= 30) {
                b0 += instAmt;
              } else if (diffDays <= 60) {
                b31 += instAmt;
              } else if (diffDays <= 90) {
                b61 += instAmt;
              } else {
                b91 += instAmt;
              }
            } else {
              b0 += instAmt; 
              remainingBal -= instAmt;
              break;
            }

            remainingBal -= instAmt;
            m++;
            if (m > 60) break;
          }
        } else if (outstandingBal > 0) {
          b0 = outstandingBal;
        }

        // Only add to installment/aging totals if balance outstanding > 0
        if (outstandingBal > 0) {
          tMonthlyInst += monthlyInst;
          t0_30 += b0;
          t31_60 += b31;
          t61_90 += b61;
          t91_plus += b91;
        }

        currentRenderedRows.push({
          totalSales: row.totalSales,
          totalReceived: row.totalReceived,
          balance: row.balance,
          monthlyInst: monthlyInst,
          b0: b0,
          b31: b31,
          b61: b61,
          b91: b91
        });

        let tr = document.createElement('tr');
        tr.innerHTML = `
          <td><span class="clickable-acc" onclick="openLedger('${row.accNo}')">${row.accNo}</span></td>
          <td><b>${row.accTitle}</b></td>
          <td>${row.salesman}</td>
          <td>${row.subSalesman}</td>
          <td>${row.mobile}</td>
          <td class="ellipsis" title="${row.product}">${row.product}</td>
          <td>${row.plan}</td>
          <td class="number">${row.totalSales.toLocaleString()}</td>
          <td class="number">${row.totalReceived.toLocaleString()}</td>
          <td class="number" style="color: #d93025;"><b>${row.balance.toLocaleString()}</b></td>

    let ledgerTimeout;

    function openLedger(accNo) {
      let modal = document.getElementById('ledgerModal');
      let content = document.getElementById('ledgerBodyContent');
      
      modal.style.display = 'block';
      content.innerHTML = '<div style="text-align:center; padding:30px; font-weight:bold; color:#1a73e8;">Loading Customer Ledger...</div>';

      if (ledgerTimeout) clearTimeout(ledgerTimeout);

      ledgerTimeout = setTimeout(function() {
        if (content.innerHTML.includes('Loading Customer Ledger')) {
          content.innerHTML = '<div style="padding: 20px; color: red; text-align: center;"><strong>Request Timeout:</strong> Server took too long. Please close and try again.</div>';
        }
      }, 10000);

      setTimeout(function() {
        google.script.run
          .withSuccessHandler(function(ledger) {
            clearTimeout(ledgerTimeout);
            if (!ledger) {
              content.innerHTML = '<p style="color:red; text-align:center;">No data returned from server.</p>';
              return;
            }
            if (ledger.error) {
              content.innerHTML = '<div style="padding: 20px; color: red; text-align: center;"><strong>Script Error:</strong> ' + ledger.message + '</div>';
              return;
            }
            renderLedgerHtml(ledger);
          })
          .withFailureHandler(function(err) {
            clearTimeout(ledgerTimeout);
            content.innerHTML = '<div style="padding: 20px; color: red; text-align: center;"><strong>Connection Error:</strong><br>' + (err.message || err) + '</div>';
          })
          .getCustomerLedger(accNo);
      }, 50);
    }

    function closeLedgerModal() {
      if (ledgerTimeout) clearTimeout(ledgerTimeout);
      document.getElementById('ledgerModal').style.display = 'none';
      document.getElementById('ledgerBodyContent').innerHTML = '';
    }

    function makeColumnsResizable() {
      const table = document.getElementById("reportTable");
      if (!table) return;

      const headers = table.querySelectorAll("th");

      headers.forEach((th, index) => {
        if (th.querySelector(".resize-handle")) return;

        const handle = document.createElement("div");
        handle.className = "resize-handle";

        handle.style.position = "absolute";
        handle.style.top = "0";
        handle.style.right = "0";
        handle.style.width = "6px";
        handle.style.height = "100%";
        handle.style.cursor = "col-resize";
        handle.style.userSelect = "none";

        th.style.position = "relative";
        th.appendChild(handle);

        let startX, startWidth;

        handle.addEventListener("mousedown", function (e) {
          startX = e.pageX;
          startWidth = th.offsetWidth;

          function doDrag(e2) {
            let newWidth = startWidth + (e2.pageX - startX);
            if (newWidth < 25) newWidth = 25;

            table.querySelectorAll("tr").forEach(row => {
              if (row.cells[index]) {
                row.cells[index].style.width = newWidth + "px";
                row.cells[index].style.minWidth = newWidth + "px";
              }
            });
          }

          function stopDrag() {
            localStorage.setItem("aging_col_" + index, th.offsetWidth);
            document.removeEventListener("mousemove", doDrag);
            document.removeEventListener("mouseup", stopDrag);
          }

          document.addEventListener("mousemove", doDrag);
          document.addEventListener("mouseup", stopDrag);
        });

        let saved = localStorage.getItem("aging_col_" + index);
        if (saved) {
          table.querySelectorAll("tr").forEach(row => {
            if (row.cells[index]) {
              row.cells[index].style.width = saved + "px";
            }
          });
        }
      });
    }
