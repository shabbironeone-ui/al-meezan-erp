  <script>
    document.getElementById('date').valueAsDate = new Date();
    updateMinMonth();

    function updateMinMonth() {
      let dateVal = document.getElementById('date').value;
      if (!dateVal) return;

      let selectedDate = new Date(dateVal);
      let currentMonthIndex = selectedDate.getMonth();

      let monthSelect = document.getElementById('nextInstallmentMonth');
      let options = monthSelect.options;
      let firstValidValue = "";

      for (let i = 0; i < options.length; i++) {
        if (i < currentMonthIndex) {
          options[i].disabled = true;
          options[i].style.color = "#ccc";
        } else {
          options[i].disabled = false;
          options[i].style.color = "#000";
          if (!firstValidValue) {
            firstValidValue = options[i].value;
          }
        }
      }

      if (monthSelect.selectedIndex < currentMonthIndex) {
        monthSelect.value = firstValidValue;
      }
    }

    function handleAccountKey(e) {
      if (e.key === 'Enter') {
        e.preventDefault();
        let accNo = document.getElementById('accountNo').value.trim();
        if (!accNo) return;

        document.getElementById('loader').style.display = 'inline-block';

        google.script.run
          .withSuccessHandler(function(res) {
            document.getElementById('loader').style.display = 'none';

            if (res.success) {
              document.getElementById('accountTitle').value = res.accountTitle;
              document.getElementById('salesman').value = res.salesman;
              document.getElementById('subSalesman').value = res.subSalesman;
              
              document.getElementById('companyName').focus();
            } else {
              alert(res.error);
              document.getElementById('accountTitle').value = '';
              document.getElementById('salesman').value = '';
              document.getElementById('subSalesman').value = '';
            }
          })
          .withFailureHandler(function(err) {
            document.getElementById('loader').style.display = 'none';
            alert("Error: " + err.message);
          })
          .getCustomerDetailsForSales(accNo); // <-- Yahan function name theek kar diya hai
      }
    }

    function calculateTotals() {
      let qty = Number(document.getElementById('qty').value) || 0;
      let salesRate = Number(document.getElementById('salesRate').value) || 0;
      let processingFee = Number(document.getElementById('processingFee').value) || 0;
      let advance = Number(document.getElementById('advance').value) || 0;
      let planMonths = Number(document.getElementById('planMonths').value) || 0;
      let firstInst = Number(document.getElementById('firstInstallment').value) || 0;

      let salesAmt = qty * salesRate;
      document.getElementById('salesAmt').value = salesAmt;

      let totalAmount = salesAmt + processingFee;
      document.getElementById('totalAmount').value = totalAmount;

      let perMonthInst = 0;
      if (planMonths > 0) {
        let balanceForInst = salesAmt - advance;
        if (balanceForInst > 0) {
          perMonthInst = Math.round(balanceForInst / planMonths);
        }
      }
      document.getElementById('installmentPerMonth').value = perMonthInst;

      let totalReceived = processingFee + advance + firstInst;
      document.getElementById('totalReceivedAtStart').value = totalReceived;
    }

    function handleFormSubmit(e) {
      e.preventDefault();
      
      let saveBtn = document.getElementById('saveBtn');
      let saveLoader = document.getElementById('saveLoader');
      saveBtn.disabled = true;
      saveLoader.style.display = 'inline-block';
      
      let formData = {
        date: document.getElementById('date').value,
        accountNo: document.getElementById('accountNo').value,
        accountTitle: document.getElementById('accountTitle').value,
        salesman: document.getElementById('salesman').value,
        subSalesman: document.getElementById('subSalesman').value,
        companyName: document.getElementById('companyName').value,
        productCategory: document.getElementById('productCategory').value,
        model: document.getElementById('model').value,
        qty: document.getElementById('qty').value,
        salesRate: document.getElementById('salesRate').value,
        salesAmt: document.getElementById('salesAmt').value,
        planMonths: document.getElementById('planMonths').value,
        advance: document.getElementById('advance').value,
        installmentPerMonth: document.getElementById('installmentPerMonth').value,
        processingFee: document.getElementById('processingFee').value,
        totalAmount: document.getElementById('totalAmount').value,
        firstInstallment: document.getElementById('firstInstallment').value,
        nextInstallmentMonth: document.getElementById('nextInstallmentMonth').value,
        totalReceivedAtStart: document.getElementById('totalReceivedAtStart').value,
        receiptNo: document.getElementById('receiptNo').value,
        remarks: document.getElementById('remarks').value
      };

      google.script.run
        .withSuccessHandler(function(res) {
          saveBtn.disabled = false;
          saveLoader.style.display = 'none';

          if (res.success) {
            alert("Sales & Receipt saved successfully!");
            document.getElementById('salesForm').reset();
            document.getElementById('date').valueAsDate = new Date();
            updateMinMonth();
            document.getElementById('accountNo').focus();
          } else {
            alert("Error: " + res.error);
          }
        })
        .withFailureHandler(function(err) {
          saveBtn.disabled = false;
          saveLoader.style.display = 'none';
          alert("Error: " + err.message);
        })
        .saveSalesEntry(formData);
    }
