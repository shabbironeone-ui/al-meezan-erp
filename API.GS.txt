/*************************************************
 * UI DIALOG TRIGGERS (FROM API)
 *************************************************/
function showReceiptForm() {
  const html = HtmlService.createHtmlOutputFromFile('ReceiptForm')
    .setWidth(520)
    .setHeight(700);
  SpreadsheetApp.getUi().showModalDialog(html, 'Daily Collection & Receipt Entry');
}

function showReportsDialog() {
  SpreadsheetApp.getUi().showModelessDialog(
    HtmlService.createHtmlOutputFromFile("ReportsView")
      .setWidth(800)
      .setHeight(560),
    "Daily Recovery & Receipts Report"
  );
}
