  <script>
    let globalData = null; // Browser-level session cache

    function fetchAndFilterData() {
      // Agar data pehle se browser mein cached hai, toh server par dobara request mat bhejo! (Instant speed)
      if (globalData !== null) {
        applySearchFilter();
        return;
      }

      // Pehli dafa server se fetch hoga
      document.getElementById('loadingContainer').style.display = 'block';
      document.getElementById('loadingMsg').innerText = "Fetching records... Please wait.";

      google.script.run
        .withSuccessHandler(function(response) {
          document.getElementById('loadingContainer').style.display = 'none';
          if (response && response.success) {
            globalData = response.data; // Save to browser session cache
            applySearchFilter();
          } else {
            alert("Error: " + (response ? response.error : "Unknown error"));
          }
        })
        .withFailureHandler(function(err) {
          document.getElementById('loadingContainer').style.display = 'none';
          alert("Script Failure: " + err.message);
        })
        .getAllReceiptsForReport();
    }

    function applySearchFilter() {
      if (!globalData) return;

      let fromDate = document.getElementById('fromDate').value;
      let toDate = document.getElementById('toDate').value;
      let accQuery = document.getElementById('searchAccInput').value.toLowerCase().trim();
      let receiptQuery = document.getElementById('searchReceiptInput').value.toLowerCase().trim();

      let fTime = fromDate ? new Date(fromDate).setHours(0,0,0,0) : null;
      let tTime = toDate ? new Date(toDate).setHours(23,59,59,999) : null;
      
      let filtered = globalData.filter(row => {
        // Date Range Check
        if (fTime && row.dateObj && row.dateObj < fTime) return false;
        if (tTime && row.dateObj && row.dateObj > tTime) return false;

        let acc = String(row.accountNo).toLowerCase();
        let title = String(row.accountTitle).toLowerCase();
        let rct = String(row.receiptNo).toLowerCase();
        let book = String(row.bookNo).toLowerCase();
        
        // Account/Name Filter Check
        let matchesAcc = true;
        if (accQuery) {
          matchesAcc = acc.includes(accQuery) || title.includes(accQuery);
        }

        // Receipt/Book Filter Check
        let matchesReceipt = true;
        if (receiptQuery) {
          matchesReceipt = rct.includes(receiptQuery) || book.includes(receiptQuery);
        }

        return matchesAcc && matchesReceipt;
      });
      
      renderTableRows(filtered);
    }

    function renderTableRows(data) {
      let tbody = document.getElementById('tableBody');
      tbody.innerHTML = "";

      if (!data || data.length === 0) {
        tbody.innerHTML = `<tr><td colspan="7" style="text-align: center; color: #777;">No records found.</td></tr>`;
        document.getElementById('totalRecords').innerText = "Total Records: 0";
        document.getElementById('totalAmount').innerText = "Total Recovered Amount: Rs. 0";
        return;
      }

      let totalRecs = data.length;
      let totalAmt = 0;
      let html = "";

      data.forEach(row => {
        totalAmt += Number(row.amount) || 0;
        html += `<tr>
          <td>${row.dateStr || ''}</td>
          <td>${row.receiptNo || ''}</td>
          <td>${row.accountNo || ''}</td>
          <td>${row.accountTitle || ''}</td>
          <td>${row.salesman || ''}</td>
          <td>${row.bookNo || ''}</td>
          <td>${(Number(row.amount) || 0).toLocaleString()}</td>
        </tr>`;
      });

      tbody.innerHTML = html;
      document.getElementById('totalRecords').innerText = "Total Records: " + totalRecs;
      document.getElementById('totalAmount').innerText = "Total Recovered Amount: Rs. " + totalAmt.toLocaleString();
    }
  
document.addEventListener('DOMContentLoaded',()=>['searchAccInput','searchReceiptInput','fromDate','toDate'].forEach(id=>document.getElementById(id).addEventListener('keypress',e=>{if(e.key==='Enter')fetchAndFilterData();})));
function resetFilters(){fromDate.value='';toDate.value='';searchAccInput.value='';searchReceiptInput.value='';if(globalData)renderTableRows(globalData);}
function printReport(){window.print();}
function openReceiptForm(){google.script.run.showReceiptForm();}
