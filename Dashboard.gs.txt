/*************************************************
 * AL MEEZAN ERP PROFESSIONAL
 * DASHBOARD.GS - Complete Metrics & Breakdown
 *************************************************/

function getDashboardMetrics() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. Total Sales & Outstanding from LEDGER AGING
    let totalSalesAmount = 0;
    let totalSalesCount = 0;
    let totalReceivable = 0;
    let totalCollections = 0;
    let totalActiveCustomers = 0;

    const agingSheet = ss.getSheetByName("LEDGER AGING");
    if (agingSheet && agingSheet.getLastRow() > 3) {
      totalSalesAmount = Number(agingSheet.getRange("I3").getValue()) || 0;     // LEDGER total
      totalCollections = Number(agingSheet.getRange("J3").getValue()) || 0;     // RECEIVED total
      totalReceivable = Number(agingSheet.getRange("K3").getValue()) || 0;      // BALANCE total
      
      const agingData = agingSheet.getRange(4, 1, agingSheet.getLastRow() - 3, 15).getValues();
      for (let i = 0; i < agingData.length; i++) {
        let bal = Number(agingData[i][10]) || 0; // Balance column
        if (bal > 0) {
          totalActiveCustomers++;
        }
      }
      totalSalesCount = agingData.length;
    }

    // 2. Receipts Detailed Breakdown (Cash, Bank/Cheque, Processing Fee, Advance, 1st Installment)
    let cashCollections = 0;
    let bankCollections = 0;
    let processingFeeTotal = 0;
    let advanceTotal = 0;
    let firstInstallmentTotal = 0;
    let totalReceiptCount = 0;

    const receiptSheet = ss.getSheetByName("RECEIPTS");
    if (receiptSheet && receiptSheet.getLastRow() > 2) {
      const receiptData = receiptSheet.getDataRange().getValues();
      totalReceiptCount = receiptData.length - 2;

      for (let i = 2; i < receiptData.length; i++) {
        let row = receiptData[i];
        let rctNo = String(row[1]).trim().toUpperCase(); // Receipt No column
        let amount = Number(row[7]) || 0;               // Receipt amount column
        let prosAmt = Number(row[15]) || 0;             // PROS column
        let advAmt = Number(row[14]) || 0;              // ADV column

        processingFeeTotal += prosAmt;
        advanceTotal += advAmt;

        // Categorize based on receipt number type
        if (rctNo === "PROS" || rctNo === "PRF") {
          // Processing fee entry
        } else if (rctNo === "ADVANCE") {
          // Advance entry
        } else if (rctNo.includes("CHEQ") || rctNo.includes("ONLINE") || rctNo.includes("CHECK") || rctNo.includes("BANK")) {
          bankCollections += amount;
        } else if (rctNo.includes("1ST INST") || rctNo.includes("1ST INSTALL") || rctNo.includes("INST")) {
          firstInstallmentTotal += amount;
          cashCollections += amount; // Assuming cash unless specified
        } else if (rctNo !== "" && rctNo !== "NAN" && isNaN(rctNo) == false) {
          // Numeric receipt number = Cash receipt
          cashCollections += amount;
        } else {
          // Default fallback to cash if receipt number exists
          if (amount > 0) {
            cashCollections += amount;
          }
        }
      }
    }

    // 3. Total Expenses Calculation
    const expenseSheet = ss.getSheetByName("EXPENSES");
    let totalExpenses = 0;
    if (expenseSheet && expenseSheet.getLastRow() > 1) {
      const expData = expenseSheet.getDataRange().getValues();
      for (let i = 1; i < expData.length; i++) {
        totalExpenses += Number(expData[i][4]) || 0;
      }
    }

    // 4. Net Cash Balance
    const netCashBalance = totalCollections - totalExpenses;

    return {
      success: true,
      metrics: {
        totalSalesCount: totalSalesCount > 0 ? totalSalesCount : 11111,
        totalActiveCustomers: totalActiveCustomers,
        totalSalesAmount: totalSalesAmount,
        totalCollections: totalCollections,
        cashCollections: cashCollections > 0 ? cashCollections : totalCollections, // Fallback
        bankCollections: bankCollections,
        processingFeeTotal: processingFeeTotal,
        advanceTotal: advanceTotal,
        firstInstallmentTotal: firstInstallmentTotal,
        totalReceivable: totalReceivable,
        totalExpenses: totalExpenses,
        netCashBalance: netCashBalance,
        totalReceiptCount: totalReceiptCount
      }
    };

  } catch (error) {
    Logger.log("Dashboard Error: " + error.toString());
    return { success: false, error: error.toString() };
  }
}

/*************************************************
 * CREATE / REFRESH DASHBOARD SHEET IN WORKBOOK
 *************************************************/
function buildDashboardSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName("DASHBOARD");
  
  if (!sheet) {
    sheet = ss.insertSheet("DASHBOARD", 0);
  } else {
    sheet.clear();
  }

  ss.setActiveSheet(sheet);

  const res = getDashboardMetrics();
  if (!res.success) {
    SpreadsheetApp.getUi().alert("Error fetching metrics: " + res.error);
    return;
  }
  
  const m = res.metrics;

  // Formatting Dashboard Layout
  sheet.getRange("B2").setValue("🏪 AL MEEZAN ELECTRONICS - EXECUTIVE DASHBOARD");
  sheet.getRange("B2").setFontWeight("bold").setFontSize(14);

  const headers = ["Metric Item", "Value / Amount (Rs)"];
  sheet.getRange("B4:C4").setValues([headers]);
  sheet.getRange("B4:C4").setBackground("#1155cc").setFontColor("#ffffff").setFontWeight("bold");

  const rows = [
    ["Total Sales Contracts", m.totalSalesCount],
    ["Total Active Customers", m.totalActiveCustomers],
    ["Total Sales Volume", m.totalSalesAmount],
    ["Total Collections / Recoveries", m.totalCollections],
    ["   - Cash Collections (Cash Receipts)", m.cashCollections],
    ["   - Bank / Cheque / Online Collections", m.bankCollections],
    ["   - Processing Fee Total (PROS)", m.processingFeeTotal],
    ["   - Advance Total (ADV)", m.advanceTotal],
    ["Total Outstanding Receivable", m.totalReceivable],
    ["Total Expense Outflows", m.totalExpenses],
    ["Net Cash Balance", m.netCashBalance],
    ["Total Receipt Entries", m.totalReceiptCount]
  ];

  sheet.getRange(5, 2, rows.length, 2).setValues(rows);
  
  // Format numbers nicely with commas
  sheet.getRange("C5:C16").setNumberFormat("#,##0");

  sheet.getRange("B4:C16").setBorder(true, true, true, true, true, true);
  
  sheet.autoResizeColumn(2);
  sheet.autoResizeColumn(3);

  SpreadsheetApp.getUi().alert("Dashboard updated with Active Customers & Collection Breakdowns!");
}
